# Обработчик команды /start
@dp.message(Command("start"))
async def start_command(message: types.Message):
    user_id = message.from_user.id
    if user_id not in user_states:
        user_states[user_id] = {"ready_to_work": False, "screenshot_count": 0, "start_time": None}
    await message.answer("🔵 Нажмите 'Готов работать', чтобы начать отправлять скриншоты.", reply_markup=ready_keyboard)

# Обработчик кнопки "Готов работать"
@dp.message(lambda message: message.text and message.text.strip() == "✅ Готов работать")
async def ready_to_work(message: types.Message):
    user_id = message.from_user.id
    user_states[user_id]["ready_to_work"] = True
    user_states[user_id]["screenshot_count"] = 0  # Сбрасываем счетчик скриншотов при новом начале
    await message.answer("✅ Можете отправлять скриншоты. После каждого скриншота я проверю его уникальность и посчитаю!")

# Обработчик кнопки "Инструкция"
@dp.message(lambda message: message.text and message.text.strip() == "ℹ️ Инструкция")

# Обработчик скриншотов
@dp.message(lambda message: message.photo)
async def handle_screenshots(message: types.Message):
    user_id = message.from_user.id
    if not user_states.get(user_id, {}).get("ready_to_work", False):
        await message.answer("🔴 Вы ещё не готовы работать. Пожалуйста, нажмите 'Готов работать'.")
        return

    # Получаем файл
    photo = message.photo[-1]  # Берём фото с максимальным размером
    file = await bot.get_file(photo.file_id)
    file_path = file.file_path

    # Скачиваем файл
    file_data = await bot.download_file(file_path)

    # Вычисляем хэш файла
    photo_hash = hashlib.md5(file_data.read()).hexdigest()

    # Проверяем на уникальность
    if photo_hash in photo_hashes:
        await message.answer("❌ Этот скриншот уже был отправлен. Отправьте другой.")
    else:
        # Добавляем хэш в общий набор
        photo_hashes.add(photo_hash)

        # Увеличиваем счетчик уникальных скриншотов для пользователя
        user_states[user_id]["screenshot_count"] += 1
        count = user_states[user_id]["screenshot_count"]

        # Если это первый скриншот, начинаем отсчёт времени
        if user_states[user_id]["start_time"] is None:
            user_states[user_id]["start_time"] = time.time()  # Запоминаем время отправки первого скрина

        # Вычисляем оставшееся время до зарплаты (7 дней)
        time_elapsed = time.time() - user_states[user_id]["start_time"]
        remaining_time = 604800 - time_elapsed  # 7 дней = 604800 секунд

        # Если время прошло, сбрасываем отсчёт
        if remaining_time <= 0:
            user_states[user_id]["start_time"] = time.time()  # Сброс времени
            remaining_time = 604800  # Устанавливаем новый отсчёт

        days_remaining = int(remaining_time // 86400)
        hours_remaining = int((remaining_time % 86400) // 3600)
        minutes_remaining = int((remaining_time % 3600) // 60)
        seconds_remaining = int(remaining_time % 60)

        # Отправляем ответ с количеством скриншотов и оставшимся временем
        await message.answer(f"📸 Скриншот засчитан! Вы отправили уже {count} скриншет(ов). Продолжайте!\n"
                             f"⏳ До зарплаты осталось {days_remaining} дней, {hours_remaining} часов, "
                             f"{minutes_remaining} минут и {seconds_remaining} секунд.\n"
                             f"💰 Минимум 1000 скриншотов для получения выплаты.\n"
                             f"⚠️ Без подписки на спонсоров выплаты не будет!")
